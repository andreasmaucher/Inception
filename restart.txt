Stop rootless Docker and point it to /goinfre:

systemctl --user stop docker

mkdir -p /goinfre/$USER/docker-root /goinfre/$USER/.config/docker
printf '{\n  "data-root": "/goinfre/%s/docker-root"\n}\n' "$USER" > /goinfre/$USER/.config/docker/daemon.json

# Make sure Docker reads that config (symlink your user config to goinfre)
rm -rf ~/.config/docker
ln -s /goinfre/$USER/.config/docker ~/.config/docker

systemctl --user daemon-reload
systemctl --user start docker
sleep 2
docker system info | grep -i 'Docker Root Dir'
# should show: /goinfre/<user>/docker-root

Purge all Docker state and caches (safe if you don’t need other local images)

# Remove any leftover content in the old small root (ignore errors if absent)
rm -rf ~/.local/share/docker/* 2>/dev/null || true

docker builder prune -af
docker buildx prune -af
docker system prune -af --volumes
docker volume prune -f
docker network prune -f

make all again
